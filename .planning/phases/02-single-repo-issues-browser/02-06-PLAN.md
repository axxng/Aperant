Create `apps/web/src/client/components/IssueDetailPanel.tsx`.

The panel uses CSS transform (not Radix Sheet) for the slide-in effect. It is `position: fixed` on the right side, covering 40% viewport width on desktop.

```typescript
import { useTranslation } from 'react-i18next';
import ReactMarkdown from 'react-markdown';
import remarkGfm from 'remark-gfm';
import { ExternalLink } from 'lucide-react';
import { Badge } from './ui/badge';
import { Button } from './ui/button';
import { ScrollArea } from './ui/scroll-area';
import { cn } from '../lib/utils';
import type { GitHubIssue } from '@shared/types/github';

interface IssueDetailPanelProps {
  issue: GitHubIssue | null;
  isOpen: boolean;
}

export function IssueDetailPanel({ issue, isOpen }: IssueDetailPanelProps) {
  const { t } = useTranslation('issues');

  const formattedDate = issue
    ? new Date(issue.createdAt).toLocaleDateString(undefined, {
        year: 'numeric',
        month: 'short',
        day: 'numeric',
      })
    : '';

  return (
    <div
      className={cn(
        'fixed right-0 top-0 h-full w-[40%] min-w-[320px] bg-background border-l border-border shadow-xl',
        'transform transition-transform duration-200 ease-out z-50',
        isOpen && issue ? 'translate-x-0' : 'translate-x-full'
      )}
    >
      {issue && (
        <ScrollArea className="h-full">
          <div className="p-6 space-y-4">
            {/* Header: title + number + state badge */}
            <div className="space-y-2">
              <div className="flex items-start gap-2 flex-wrap">
                <Badge
                  variant={issue.state === 'open' ? 'success' : 'muted'}
                  className="flex-shrink-0 mt-0.5"
                >
                  {t(`state.${issue.state}`)}
                </Badge>
                <span className="text-xs text-muted-foreground flex-shrink-0 mt-1">
                  #{issue.number}
                </span>
              </div>
              <h2 className="text-base font-semibold leading-tight">{issue.title}</h2>
            </div>

            {/* Meta: labels + assignees + created date */}
            <div className="space-y-2">
              {/* Labels — full name with color background */}
              {issue.labels.length > 0 && (
                <div className="flex flex-wrap gap-1.5">
                  {issue.labels.map(label => (
                    <span
                      key={label.name}
                      className="inline-flex items-center gap-1 text-xs px-2 py-0.5 rounded-full border"
                      style={{
                        backgroundColor: `#${label.color}20`,
                        borderColor: `#${label.color}60`,
                        color: `#${label.color}`,
                      }}
                    >
                      {label.name}
                    </span>
                  ))}
                </div>
              )}

              {/* Assignees — avatar + login */}
              {issue.assignees.length > 0 && (
                <div className="flex flex-wrap gap-2">
                  {issue.assignees.map(assignee => (
                    <div key={assignee.login} className="flex items-center gap-1.5">
                      <div className="h-5 w-5 rounded-full bg-muted flex items-center justify-center overflow-hidden flex-shrink-0">
                        {assignee.avatarUrl ? (
                          <img
                            src={assignee.avatarUrl}
                            alt={assignee.login}
                            className="h-full w-full object-cover"
                          />
                        ) : (
                          <span className="text-[9px] font-medium text-muted-foreground uppercase">
                            {assignee.login.slice(0, 2)}
                          </span>
                        )}
                      </div>
                      <span className="text-xs text-muted-foreground">{assignee.login}</span>
                    </div>
                  ))}
                </div>
              )}

              {/* Created date */}
              <p className="text-xs text-muted-foreground">{formattedDate}</p>
            </div>

            {/* View on GitHub link — BROWSE-08 */}
            <Button variant="outline" size="sm" asChild>
              <a
                href={issue.htmlUrl}
                target="_blank"
                rel="noreferrer"
                className="inline-flex items-center gap-1.5"
              >
                <ExternalLink className="h-3.5 w-3.5" />
                {t('detail.viewOnGitHub')}
              </a>
            </Button>

            {/* Divider */}
            <div className="border-t border-border" />

            {/* Markdown body — react-markdown + remark-gfm (D-17) */}
            {issue.body ? (
              <ReactMarkdown
                remarkPlugins={[remarkGfm]}
                className="prose prose-sm dark:prose-invert max-w-none"
              >
                {issue.body}
              </ReactMarkdown>
            ) : (
              <p className="text-sm text-muted-foreground italic">{t('detail.noBody', 'No description provided.')}</p>
            )}
          </div>
        </ScrollArea>
      )}
    </div>
  );
}
```

Critical correctness rules:
1. `import remarkGfm from 'remark-gfm'` — DEFAULT import. Do NOT use named import `{ remarkGfm }` — `remark-gfm` is ESM-only and uses `export default`.
2. `remarkPlugins={[remarkGfm]}` — wrap in array; plain function reference.
3. `#${label.color}` prefix on ALL color properties (backgroundColor, borderColor, color).
4. `target="_blank" rel="noreferrer"` on the View on GitHub anchor — prevents opener relationship.
5. `issue.htmlUrl` (not `issue.url`) — `url` is the API URL; `htmlUrl` is the github.com page URL.
6. Panel is `position: fixed` — it overlays the list panel; `z-50` keeps it above everything.
7. The `className` prop on `<ReactMarkdown>` applies to the wrapper element; `prose prose-sm dark:prose-invert max-w-none` applies only if `@tailwindcss/typography` was successfully registered in Wave 0. If typography plugin failed (fallback comment in globals.css), replace with `text-sm leading-relaxed space-y-3` — this minimal fallback is readable without the typography plugin.

For the `detail.noBody` translation key: the key `noBody` needs to be added to en/issues.json and fr/issues.json. If it is not already present, add it in this task (it was omitted from the 02-03 plan's key list but is needed here):

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| GitHub API → issue.body → react-markdown | Issue body Markdown string rendered to React nodes by react-markdown |
| GitHub API → issue.htmlUrl → anchor href | htmlUrl used as href for "View on GitHub" link |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-06-01 | Tampering (XSS) | issue.body Markdown rendering | mitigate | `react-markdown` outputs React nodes — no `dangerouslySetInnerHTML`, no `innerHTML` assignment; inherently safe per D-17; ASVS V5.3 satisfied |
| T-02-06-02 | Tampering | issue.htmlUrl as anchor href | accept | URL comes from GitHub API `html_url` field; GitHub only returns `https://github.com/...` URLs; `rel="noreferrer"` prevents opener relationship; `target="_blank"` opens in new context without JS access to the originating page |
| T-02-06-03 | Tampering | Label color CSS values | accept | Same as T-02-04-02 and T-02-05-03; CSS property injection via label.color is low severity; no JS execution vector |
| T-02-06-04 | Information Disclosure | Issue body visible in panel | accept | Panel is only rendered when user clicks an issue they can already see in the list; no new information disclosure beyond what exists at the API level |
</threat_model>
