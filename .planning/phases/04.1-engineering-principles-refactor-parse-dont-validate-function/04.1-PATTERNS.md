export function validateOAuthState(cookieHeader: string, queryState: string | undefined): boolean {
  const cookieState = cookieHeader
    .split('; ')
    .find((c) => c.startsWith('oauth_state='))
    ?.split('=')[1];
  return !!(cookieState && queryState && cookieState === queryState);
}

export function determineRole(userCount: number): 'admin' | 'member' {
  return userCount === 0 ? 'admin' : 'member';
}

export function buildEmailFallback(email: string | null | undefined, login: string): string {
  return email || `${login}@github.invalid`;
}

export function buildUserRecord(
  githubUser: z.infer<typeof gitHubUserSchema>,
  tokenData: z.infer<typeof oauthTokenResponseSchema>,

    githubId: String(githubUser.id),
    email: buildEmailFallback(githubUser.email, githubUser.login),
    name: githubUser.name || githubUser.login,
    githubLogin: githubUser.login,
    githubToken: tokenData.access_token,

```typescript
export default async function handler(req: VercelRequest, res: VercelResponse) {
  if (req.method !== 'GET') return res.status(405).json({ error: 'Method not allowed' });

  // 1. Parse input (state validation is part of parse — CSRF check is input validation)
  if (!validateOAuthState(req.headers.cookie || '', req.query.state as string | undefined)) {
    return res.status(400).json({ error: 'Invalid OAuth state' });
  }
  const code = req.query.code as string | undefined;
  if (!code) return res.status(400).json({ error: 'Missing OAuth code' });

  // 2. Authorize — N/A for OAuth callback (it's the auth entry point)

  try {
    await ensureDb();

    // I/O: exchange code for token
    const tokenResponse = await fetch('https://github.com/login/oauth/access_token', { ... });
    const tokenData = oauthTokenResponseSchema.parse(await tokenResponse.json()); // parse GitHub response
    if (tokenData.error || !tokenData.access_token) {
      return res.status(400).json({ error: 'OAuth token exchange failed' });
    }

    // I/O: fetch GitHub user
    const rawUser = await fetch('https://api.github.com/user', { ... }).then(r => r.json());
    const githubUser = gitHubUserSchema.parse(rawUser); // parse GitHub response

    // I/O: get user count
    const count = await userCount();

    // 3. Pure domain function(s)
    const userRecord = buildUserRecord(githubUser, tokenData, count);

    // I/O: upsert
    const user = await upsertOAuthUser(userRecord);
    const jwt = createToken(user.id, user.email, user.role);

    // 4. Respond
    const securePart = process.env.NODE_ENV === 'production' ? '; Secure' : '';
    res.setHeader('Set-Cookie', `oauth_state=; HttpOnly${securePart}; Max-Age=0; Path=/`);
    res.redirect(302, `/?token=${jwt}`);
  } catch (error: unknown) {
    console.error('[oauth callback]', error);
    res.status(500).json({ error: 'Internal server error' });
  }
}
```

### `apps/web/api/tasks/index.ts` (controller, CRUD)

**Analog:** self — this file is also the canonical reference for other handlers

**Current deviation** (lines 8–38): Auth check runs before parse for GET; POST uses `safeParse` (not `parse`) with manual 400; GET has no Zod parse of `productId` query param.

**Target 4-step shape:**
```typescript
export default async function handler(req: VercelRequest, res: VercelResponse) {
  await ensureDb();

  switch (req.method) {
    case 'GET': {
      // 1. Parse input
      const productId = req.query.productId ? z.string().uuid().parse(req.query.productId) : undefined;
      // 2. Authorize
      const user = await authenticateRequest(req, res);
      if (!user) return;
      if (!hasRole(user, 'admin', 'member')) return res.status(403).json({ error: 'Insufficient permissions' });
      // 3. Pure domain / DB call
      const tasks = productId ? await getTasksByProduct(productId) : await getAllTasks();
      // 4. Respond
      return res.json(tasks);
    }

    case 'POST': {
      // 1. Parse input — z.parse() throws ZodError → caught by outer try/catch → 500
      // Use safeParse here specifically because user input errors should be 400 not 500
      const result = createTaskSchema.safeParse(req.body);
      if (!result.success) return res.status(400).json({ error: 'Invalid input', details: result.error.flatten().fieldErrors });
      // 2. Authorize
      const user = await authenticateRequest(req, res);
      if (!user) return;
      if (!hasRole(user, 'admin', 'member')) return res.status(403).json({ error: 'Insufficient permissions' });
      // 3. Pure domain
      const task = await createTask(result.data);
      await broadcastEvent('task_created', task);
      // 4. Respond
      return res.status(201).json(task);
    }

    default:
      return res.status(405).json({ error: 'Method not allowed' });
  }
}
```

**Note on parse vs safeParse:** Per D-02, `z.parse()` (throws) is for programmer/deployment bugs (DB rows, internal data). For user-submitted API body/query params, `safeParse` + 400 is the correct pattern — it is user-recoverable. Both are valid in the 4-step shape; the difference is the error surface.

```markdown
## Engineering Principles

These four principles apply to **all code in `apps/web/`**. Future agents MUST follow them for every change.

### 1. Parse-Don't-Validate

Parse raw input into typed domain objects at every untrusted boundary. Reject immediately on failure. Never do runtime type checks (`as T`, `instanceof`, manual `if typeof`) inside domain logic.

**Untrusted boundaries:**
- API request body/query params → `schema.parse(req.body)` / `schema.safeParse()`
- DB rows → `rowSchema.parse(row)` in every `rowToXxx()` mapper
- External API responses (GitHub, OAuth) → `apiSchema.parse(await response.json())`

**On parse failure:** Use `z.parse()` (throws `ZodError`). The handler's `try/catch` surfaces a 500 for programmer/deployment bugs. For user-submitted input, use `z.safeParse()` + 400 response. Never return `null` or wrap in `Result` types.

```typescript
// CORRECT — parse at DB boundary
function rowToTask(row: unknown): Task {
  const parsed = taskDbRowSchema.parse(row); // throws ZodError on bad row
  return { id: parsed.id, status: parsed.status, ... };
}

// WRONG — blind cast
function rowToTask(row: any): Task {
  return { id: row.id as string, status: row.status as TaskStatus }; // no validation
}
```

All Zod schemas for DB rows, GitHub API responses, and OAuth responses live in `api/_lib/validation.ts`.

### 2. Functional Core / Imperative Shell

Pure functions carry business logic (no I/O, deterministic). A thin imperative shell orchestrates I/O and delegates to pure functions.

**Every API handler follows the 4-step shape without exception:**
1. Parse input (Zod schema)
2. Authorize (`authenticateRequest` + `hasRole`)
3. Call pure domain function(s)
4. Respond (`res.json` / `res.status`)

**Pure functions stay in the same file** as the handler that uses them. Separate with a clear `// === Pure domain logic ===` comment block. No new `*-logic.ts` files.

```typescript
export default async function handler(req: VercelRequest, res: VercelResponse) {
  await ensureDb();

  // 1. Parse
  const id = uuidSchema.parse(req.query.id);
  const body = updateTaskSchema.parse(req.body);

  // 2. Authorize
  const user = await authenticateRequest(req, res);
  if (!user) return;
  if (!hasRole(user, 'admin', 'member')) return res.status(403).json({ error: 'Insufficient permissions' });

  // 3. Pure domain function(s)
  const payload = buildUpdatePayload(body);

  // 4. Respond
  const task = await updateTask(id, payload);
  return res.json(task);
}

// === Pure domain logic ===
function buildUpdatePayload(input: z.infer<typeof updateTaskSchema>): UpdateTaskInput { ... }
```

### 3. FSM / Illegal State Elimination

Use discriminated unions to make invalid states unrepresentable. Every domain object with lifecycle state gets a typed state machine.

**Pattern:**
```typescript
// CORRECT — illegal state unrepresentable
export type Task =
  | (TaskBase & { status: 'backlog' })
  | (TaskBase & { status: 'error'; reviewReason: ReviewReason })  // error REQUIRES reviewReason
  | (TaskBase & { status: 'pr_created'; githubIssueNumber: number; githubIssueUrl: string; githubRepo: string });

// WRONG — all fields optional on all variants
export interface Task {
  status: TaskStatus;
  reviewReason?: ReviewReason;  // can be missing on 'error' status — silently invalid
}
```

**Current discriminated unions in `apps/web`:**
- `Task` — 8 status variants in `src/shared/types/task.ts`
- `GithubSyncState` — `idle | pending | retrying | failed | complete` in `src/shared/types/task.ts`
- `TriageState` — `untouched | prioritized | complete` in `api/_lib/db/triage.ts`
- `ProductSource` — `repo | repos | github_project | gitlab_project` in `api/_lib/validation.ts`

When adding a new domain type with lifecycle: define a discriminated union first, then write the `rowToXxx()` mapper that assembles the correct variant. Keep a separate string literal union (e.g., `TaskStatusKey`) for use in `Record<>` keys.

### 4. Red-Green TDD

Write a failing test before writing implementation for every new or refactored function. Tests live co-located next to source (`api/_lib/db/tasks.test.ts` beside `tasks.ts`).

**Test runner:** `cd apps/web && npm test` (Vitest). Individual file: `cd apps/web && npx vitest run path/to/file.test.ts`.
