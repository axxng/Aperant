import { useInfiniteQuery, useQuery } from '@tanstack/react-query';

// useInfiniteQuery for issues (BROWSE-07 — Load More):
const { data, isLoading, isFetchingNextPage, fetchNextPage, hasNextPage, error, refetch } =
  useInfiniteQuery<{ issues: GitHubIssue[]; hasMore: boolean }>({
    queryKey: ['issues', owner, repo, { state, labels: [...labels].sort(), assignee }],
    queryFn: async ({ pageParam = 1 }) => {
      const params = new URLSearchParams({ state, page: String(pageParam), per_page: '50' });
      if (labels.length > 0) params.set('labels', labels.join(','));
      if (assignee) params.set('assignee', assignee);
      const res = await fetch(`/api/github/repos/${owner}/${repo}/issues?${params}`, {
        credentials: 'include',
      });
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        throw new Error(body.error || `HTTP ${res.status}`);
      }
      return res.json();
    },
    initialPageParam: 1,
    getNextPageParam: (lastPage, allPages) =>
      lastPage.hasMore ? allPages.length + 1 : undefined,
  });

// useQuery for labels (parallel fetch — BROWSE-03):
const { data: labelsData, isError: labelsError } = useQuery<{ labels: GitHubLabel[] }>({
  queryKey: ['labels', owner, repo],
  queryFn: async () => {
    const res = await fetch(`/api/github/repos/${owner}/${repo}/labels`, {
      credentials: 'include',
    });
    if (!res.ok) throw new Error('labels fetch failed');
    return res.json();
  },
  staleTime: 5 * 60 * 1000,  // 5 minutes — labels change rarely
});

// Flatten pages for rendering:
const allIssues = data?.pages.flatMap(p => p.issues) ?? [];

```typescript
import { useState, useEffect, useMemo } from 'react';
import { useParams, NavLink } from 'react-router-dom';
import { useTranslation } from 'react-i18next';
import { useInfiniteQuery, useQuery } from '@tanstack/react-query';
import { GitBranch, AlertCircle, Loader2 } from 'lucide-react';
import { useProductStore } from '../stores/product-store';
import { useIssuesFilters } from '../hooks/useIssuesFilters';
import { IssuesFilterBar } from './IssuesFilterBar';
import { IssueListRow } from './IssueListRow';
import { IssueDetailPanel } from './IssueDetailPanel';
import { IssueSkeletonRow } from './IssueSkeletonRow';
import { Button } from './ui/button';
import { cn } from '../lib/utils';
import type { GitHubIssue } from '@shared/types/github';

interface GitHubLabel {
  id: number;
  name: string;
  color: string;
  description: string | null;
}

interface PaginatedIssuesResult {
  issues: GitHubIssue[];
  hasMore: boolean;
}

interface LabelsResult {
  labels: GitHubLabel[];
}

export function IssuesView() {
  const { productId } = useParams<{ productId: string }>();
  const { t } = useTranslation('issues');
  const { products, setActiveProduct } = useProductStore();
  const [selectedIssueId, setSelectedIssueId] = useState<number | null>(null);

  // Sync active product — mirrors ProductView pattern (Pitfall 3 avoidance)
  useEffect(() => {
    if (productId) setActiveProduct(productId);
  }, [productId, setActiveProduct]);

  // Derive owner/repo directly from products array (NOT getActiveProduct)
  const product = products.find(p => p.id === productId);
  const repoSource = product?.sources.find(s => s.type === 'repo');

  const { state, labels, assignee, search, setSearch, setFilters, resetFilters, hasActiveFilters } =
    useIssuesFilters();

  // Issues query — useInfiniteQuery for "Load More" pagination (BROWSE-07)
  // queryKey includes all filter state for correct cache invalidation (Pitfall 2)
  const {
    data,
    isLoading,
    isFetchingNextPage,
    fetchNextPage,
    hasNextPage,
    error,
    refetch,
  } = useInfiniteQuery<PaginatedIssuesResult>({
    queryKey: ['issues', repoSource?.owner, repoSource?.repo, {
      state,
      labels: [...labels].sort(),  // sort to avoid false cache misses from ordering
      assignee,
    }],
    queryFn: async ({ pageParam }) => {
      if (!repoSource) throw new Error('No repo source');
      const params = new URLSearchParams({
        state,
        page: String(pageParam ?? 1),
        per_page: '50',
      });
      if (labels.length > 0) params.set('labels', labels.join(','));
      if (assignee) params.set('assignee', assignee);
      const res = await fetch(
        `/api/github/repos/${repoSource.owner}/${repoSource.repo}/issues?${params}`,
        { credentials: 'include' }
      );
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        throw new Error(body.error || `HTTP ${res.status}`);
      }
      return res.json() as Promise<PaginatedIssuesResult>;
    },
    initialPageParam: 1,
    getNextPageParam: (lastPage, allPages) =>
      lastPage.hasMore ? allPages.length + 1 : undefined,
    enabled: Boolean(repoSource),
  });

  // Labels query — parallel to issues (D-08); staleTime=5min because labels change rarely
  const { data: labelsData, isError: labelsError } = useQuery<LabelsResult>({
    queryKey: ['labels', repoSource?.owner, repoSource?.repo],
    queryFn: async () => {
      if (!repoSource) throw new Error('No repo source');
      const res = await fetch(
        `/api/github/repos/${repoSource.owner}/${repoSource.repo}/labels`,
        { credentials: 'include' }
      );
      if (!res.ok) throw new Error('labels fetch failed');
      return res.json() as Promise<LabelsResult>;
    },
    staleTime: 5 * 60 * 1000,
    enabled: Boolean(repoSource),
  });

  // Flatten pages into single array
  const allIssues = data?.pages.flatMap(p => p.issues) ?? [];

  // Client-side title search (D-10 — no extra API calls)
  const filteredIssues = useMemo(() => {
    if (!search.trim()) return allIssues;
    const q = search.toLowerCase();
    return allIssues.filter(i => i.title.toLowerCase().includes(q));
  }, [allIssues, search]);

  // Derive assignees from loaded issues (D-09 — no separate API call)
  const availableAssignees = useMemo(() => {
    const logins = new Set(allIssues.flatMap(i => i.assignees.map(a => a.login)));
    return Array.from(logins).sort();
  }, [allIssues]);

  const selectedIssue = allIssues.find(i => i.id === selectedIssueId) ?? null;

  // No repo source — show error state
  if (!repoSource) {
    return (
      <div className="flex flex-col h-full">
        <TabSwitcher productId={productId ?? ''} t={t} />
        <div className="flex flex-1 items-center justify-center">
          <div className="text-center space-y-2">
            <AlertCircle className="h-8 w-8 text-muted-foreground mx-auto" />
            <p className="text-sm text-muted-foreground">{t('error.noRepoSource', 'This product does not have a connected GitHub repository.')}</p>
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className="flex flex-col h-full overflow-hidden">
      {/* Tab switcher: Kanban | Issues */}
      <TabSwitcher productId={productId ?? ''} t={t} />

      {/* Filter bar */}
      <IssuesFilterBar
        filters={{ state, labels, assignee }}
        search={search}
        onSearchChange={setSearch}
        onFilterChange={setFilters}
        onReset={resetFilters}
        hasActiveFilters={hasActiveFilters}
        availableLabels={labelsData?.labels ?? []}
        labelsError={labelsError}
        availableAssignees={availableAssignees}
      />

      {/* Split pane: issues list + detail panel */}
      <div className="flex flex-1 overflow-hidden">
        {/* Issues list pane */}
        <div className="flex-1 flex flex-col overflow-y-auto">
          {isLoading ? (
            // 8 skeleton rows while first page loads (D-18)
            Array.from({ length: 8 }).map((_, i) => <IssueSkeletonRow key={i} />)
          ) : error ? (
            // Error state
            <div className="flex flex-1 items-center justify-center p-8">
              <div className="text-center space-y-3">
                <AlertCircle className="h-8 w-8 text-muted-foreground mx-auto" />
                <h3 className="text-sm font-medium">{t('error.heading')}</h3>
                <p className="text-xs text-muted-foreground">{t('error.body')}</p>
                <Button variant="outline" size="sm" onClick={() => refetch()}>
                  {t('error.retry')}
                </Button>
              </div>
            </div>
          ) : filteredIssues.length === 0 ? (
            // Empty state (D-19)
            <div className="flex flex-1 items-center justify-center p-8">
              <div className="text-center space-y-3">
                <GitBranch className="h-8 w-8 text-muted-foreground mx-auto" />
                <h3 className="text-sm font-medium">
                  {state === 'open' ? t('empty.openHeading') : t('empty.closedHeading')}
                </h3>
                <p className="text-xs text-muted-foreground">{t('empty.body')}</p>
                <Button variant="outline" size="sm" asChild>
                  <a
                    href={`https://github.com/${repoSource.owner}/${repoSource.repo}/issues`}
                    target="_blank"
                    rel="noreferrer"
                  >
                    {t('empty.viewOnGitHub')}
                  </a>
                </Button>
              </div>
            </div>
          ) : (
            <>
              {filteredIssues.map(issue => (
                <IssueListRow
                  key={issue.id}
                  issue={issue}
                  isSelected={issue.id === selectedIssueId}
                  onClick={() => setSelectedIssueId(issue.id)}
                />
              ))}

              {/* Load More button (BROWSE-07) */}
              {hasNextPage && (
                <div className="flex justify-center p-4">
                  <Button
                    variant="outline"
                    onClick={() => fetchNextPage()}
                    disabled={isFetchingNextPage}
                  >
                    {isFetchingNextPage ? (
                      <>
                        <Loader2 className="h-4 w-4 mr-2 animate-spin" />
                        {t('list.loadMore')}
                      </>
                    ) : (
                      t('list.loadMore')
                    )}
                  </Button>
                </div>
              )}
            </>
          )}
        </div>

        {/* Detail panel — slide-in from right */}
        <IssueDetailPanel
          issue={selectedIssue}
          isOpen={selectedIssueId !== null}
        />
      </div>
    </div>
  );
}

// ─── Internal sub-components ────────────────────────────────────────────────

interface TabSwitcherProps {
  productId: string;
  t: (key: string) => string;
}

function TabSwitcher({ productId, t }: TabSwitcherProps) {
  return (
    <div className="flex items-center border-b border-border px-4 flex-shrink-0">
      <NavLink
        to={`/products/${productId}`}
        end
        className={({ isActive }) =>
          cn(
            'px-4 py-2 text-sm font-medium border-b-2 -mb-px transition-colors',
            isActive
              ? 'border-primary text-foreground font-semibold'
              : 'border-transparent text-muted-foreground hover:text-foreground'
          )
        }
      >
        Kanban
      </NavLink>
      <NavLink
        to={`/products/${productId}/issues`}
        className={({ isActive }) =>
          cn(
            'px-4 py-2 text-sm font-medium border-b-2 -mb-px transition-colors',
            isActive
              ? 'border-primary text-foreground font-semibold'
              : 'border-transparent text-muted-foreground hover:text-foreground'
          )
        }
      >
        {t('tab.issues')}
      </NavLink>
    </div>
  );
}
```

Critical correctness rules:
1. `queryKey: ['issues', owner, repo, { state, labels: [...labels].sort(), assignee }]` — sort labels array to avoid false cache misses from different ordering.
2. `products.find(p => p.id === productId)` — NEVER `getActiveProduct()` (Pitfall 3).
3. `useEffect(() => { if (productId) setActiveProduct(productId) }, [productId, setActiveProduct])` — sync active product so sidebar and breadcrumbs work.
4. `enabled: Boolean(repoSource)` on both queries — prevents query from running when repoSource is undefined.
5. `credentials: 'include'` on all `fetch()` calls — sends session cookies for authentication.
6. Client-side search uses `useMemo` to avoid re-filtering on every render.
7. `NavLink` with `end` prop on Kanban tab — without `end`, `/products/:id` NavLink would also match `/products/:id/issues`, making both tabs active simultaneously.
8. "Kanban" tab label: use the hardcoded string "Kanban" (not i18n) since this references the existing Kanban feature that has its own i18n namespace. Alternatively look up `t('navigation:kanban')` if that key exists. Use hardcoded "Kanban" as the safe default.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → /api/github/repos/:owner/:repo/issues | Authenticated fetch with session cookie |
| Browser → /api/github/repos/:owner/:repo/labels | Authenticated fetch with session cookie |
| GitHub API response → React rendering | Issue data rendered via text nodes and react-markdown |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-07-01 | Elevation of Privilege | fetch() calls without auth | mitigate | All fetch calls use `credentials: 'include'` to send session cookies; server-side `authenticateRequest()` validates the session; 401 propagates to TanStack Query error state |
| T-02-07-02 | Tampering | queryKey — label array ordering | accept | Labels sorted before use in queryKey to prevent false cache misses from ordering differences; no security impact, only correctness |
| T-02-07-03 | Information Disclosure | error messages from API in error state | accept | Error state shows generic i18n copy (`error.heading`, `error.body`), not the raw error message from the server; no leakage of internal error details to the UI |
| T-02-07-04 | Denial of Service | useInfiniteQuery with no fetch limit | accept | Each page is capped at 50 issues (per_page=50); Load More is manual (not infinite scroll); maximum data loaded is bounded by user action count |
| T-02-07-05 | Tampering | URL params state/labels/assignee used in fetch URL | mitigate | Params are passed to the server-side issues route; `githubIssueQuerySchema` validates state/page/per_page; labels/assignee are passed verbatim to GitHub API which applies its own validation (T-03-05 accepted in Phase 1) |
</threat_model>
