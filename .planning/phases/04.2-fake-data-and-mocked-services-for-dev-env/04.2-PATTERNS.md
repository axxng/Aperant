**Target pattern — MOCK_SERVICES gate** (replace `getClient()` body):
```typescript
export function getClient(): Client {
  if (!client) {
    if (process.env.MOCK_SERVICES === 'true') {
      client = createClient({ url: 'file:dev.db' });
    } else {
      client = createClient({
        url: process.env.TURSO_DATABASE_URL!,
        authToken: process.env.TURSO_AUTH_TOKEN,
      });
    }
  }
  return client;
}
```

**Critical constraint:** The module-level `let client: Client | null = null` singleton pattern must be preserved. All existing `vi.mock('./client.js', () => ({ getClient: () => ({ execute: mockExecute }) }))` test mocks replace the entire module export and never reach this conditional — existing tests are not affected.

```typescript
// Mock @libsql/client to control createClient behavior
const mockExecute = vi.fn().mockResolvedValue({ rows: [] });
const mockCreateClient = vi.fn().mockReturnValue({ execute: mockExecute, executeMultiple: vi.fn() });
vi.mock('@libsql/client', () => ({
  createClient: mockCreateClient,
}));

// Re-import getClient after setting up env
describe('getClient()', () => {
  afterEach(() => {
    vi.resetModules();
    delete process.env.MOCK_SERVICES;
    delete process.env.TURSO_DATABASE_URL;
  });

  it('MOCK-01: returns file:dev.db client when MOCK_SERVICES=true', async () => {
    process.env.MOCK_SERVICES = 'true';
    const { getClient } = await import('./client.js');
    getClient();
    expect(mockCreateClient).toHaveBeenCalledWith({ url: 'file:dev.db' });
  });

  it('MOCK-02: returns Turso client when MOCK_SERVICES unset', async () => {
    process.env.TURSO_DATABASE_URL = 'libsql://test.turso.io';
    const { getClient } = await import('./client.js');
    getClient();
    expect(mockCreateClient).toHaveBeenCalledWith(
      expect.objectContaining({ url: 'libsql://test.turso.io' })
    );
  });
});
```

**Note:** `vi.resetModules()` in `afterEach` is required because `client.ts` has module-level singleton state (`let client = null`). Without resetting, the cached client from the first test bleeds into subsequent tests.
