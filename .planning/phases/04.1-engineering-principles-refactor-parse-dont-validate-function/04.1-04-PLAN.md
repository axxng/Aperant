```typescript
  await ensureDb();
  if (req.method !== 'VERB') return res.status(405).json({ error: 'Method not allowed' });
```

```typescript
export function validateOAuthState(cookieHeader: string, queryState: string | undefined): boolean {
  const cookieState = cookieHeader.split('; ').find((c) => c.startsWith('oauth_state='))?.split('=')[1];
  return !!(cookieState && queryState && cookieState === queryState);
}
export function determineRole(userCount: number): 'admin' | 'member' {
  return userCount === 0 ? 'admin' : 'member';
}
export function buildEmailFallback(email: string | null | undefined, login: string): string {
  return email || `${login}@github.invalid`;
}
export function buildUserRecord(
  githubUser: z.infer<typeof gitHubUserSchema>,
  tokenData: z.infer<typeof oauthTokenResponseSchema>,
  userCount: number
): UpsertOAuthUserInput {
  return {
    githubId: String(githubUser.id),
    email: buildEmailFallback(githubUser.email, githubUser.login),
    name: githubUser.name || githubUser.login,
    githubLogin: githubUser.login,
    githubToken: tokenData.access_token,
    role: determineRole(userCount),
  };
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP request → handler (Step 1) | Untrusted path params and query strings enter the handler |
| OAuth callback → GitHub token/user | GitHub OAuth response (untrusted external) enters domain logic |
| Auth step (Step 2) | User identity crosses the trust boundary into authorized domain actions |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04.1-04-01 | Spoofing | OAuth callback CSRF check | mitigate | `validateOAuthState()` is extracted as a pure function with direct unit tests — verifies cookie matches query state before any I/O proceeds |
| T-04.1-04-02 | Elevation of Privilege | Handler auth step out of order | mitigate | 4-step shape enforces authorize BEFORE calling domain functions — parse (step 1) never accesses protected data, only validates input shape |
| T-04.1-04-03 | Tampering | `as string` path param casts bypass validation | mitigate | All path params replaced with z.string().parse() or schema.parse() — unexpected chars in owner/repo/id throw ZodError → 500 |
| T-04.1-04-04 | Tampering | updateTaskSchema still accepted githubSyncPending (removed in Plan 02) | mitigate | Already removed in Plan 02; handlers that called updateTask(id, { githubSyncPending }) are updated to use updateTaskSyncState() in this plan |
| T-04.1-04-05 | Information Disclosure | OAuth token response accessed without schema validation | mitigate | oauthTokenResponseSchema.parse(await tokenResponse.json()) — ZodError on malformed GitHub response instead of accessing undefined fields |
