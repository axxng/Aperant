    Replace githubFetch() body — store fetch result and inspect before returning:
    ```typescript
    export async function githubFetch(url: string, options: RequestInit = {}): Promise<Response> {
      const token = await getGitHubToken();
      const response = await fetch(url, {
        ...options,
        headers: {
          'Accept': 'application/vnd.github.v3+json',
          'Authorization': `Bearer ${token}`,
          'X-GitHub-Api-Version': '2022-11-28',
          ...options.headers,
        },
      });

      // Rate-limit detection: primary (429 or 403 + remaining=0) and secondary (retry-after header)
      const isRateLimitStatus = response.status === 429 || response.status === 403;
      if (isRateLimitStatus) {
        const remaining = response.headers.get('x-ratelimit-remaining');
        const retryAfterHeader = response.headers.get('retry-after');   // secondary: already in seconds
        const resetHeader = response.headers.get('x-ratelimit-reset');  // primary: unix timestamp

        const isPrimaryLimit = remaining === '0';
        const isSecondaryLimit = retryAfterHeader !== null;

        if (isPrimaryLimit || isSecondaryLimit) {
          let retryAfter: number;
          if (retryAfterHeader !== null) {
            // Secondary rate limit: retry-after is in seconds
            retryAfter = Math.max(0, parseInt(retryAfterHeader, 10));
          } else if (resetHeader !== null) {
            // Primary rate limit: x-ratelimit-reset is a Unix timestamp
            retryAfter = Math.max(0, Math.ceil(parseInt(resetHeader, 10) - Date.now() / 1000));
          } else {
            retryAfter = 60; // fallback per D-02
          }
          throw new GitHubRateLimitError(retryAfter);
        }
      }

      return response;
    }
    ```

| Boundary | Description |
|----------|-------------|
| githubFetch → GitHub API | Response headers come from GitHub's servers; retryAfter value is derived from them |
| Client → githubFetch | Callers pass URL and options; URL is constructed by route handlers using encodeURIComponent |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Tampering | `retryAfter` header parsing | accept | Headers originate from GitHub API response, not user input; value clamped to Math.max(0, ...) preventing negative numbers |
| T-01-02 | Information Disclosure | `GitHubRateLimitError.message` | accept | Message only contains public retry timing info; no credentials or internal state leaked |
| T-01-03 | Denial of Service | Malformed `retry-after` header | mitigate | `parseInt(..., 10)` returns NaN on malformed input; `Math.max(0, NaN)` = 0, which is a safe fallback; callers get retryAfter=0 |
</threat_model>
