    ```typescript
    import type { VercelRequest, VercelResponse } from '@vercel/node';
    import { z } from 'zod';
    import { ensureDb } from '../../../_lib/db/client.js';
    import { authenticateRequest } from '../../../_lib/auth/middleware.js';
    import { getTriageRecord, upsertTriageRecord } from '../../../_lib/db/triage.js';
    import { githubOwnerRepoSchema } from '../../../_lib/validation.js';

    // Validate path parameters (D-06: auth required, path must be valid)
    const triagePathSchema = githubOwnerRepoSchema.extend({
      number: z.string().regex(/^\d+$/, 'Issue number must be a positive integer'),
    });

    // PUT body schema — accepts any subset of { isTriaged, priority } per D-05
    const triagePutBodySchema = z.object({
      isTriaged: z.boolean().optional(),
      priority: z.enum(['critical', 'high', 'medium', 'low']).nullable().optional(),
    });

    export default async function handler(req: VercelRequest, res: VercelResponse) {
      await ensureDb();

      // D-06: require authentication on both GET and PUT
      const user = await authenticateRequest(req, res);
      if (!user) return;

      // Validate path params
      const pathResult = triagePathSchema.safeParse({
        owner: req.query.owner,
        repo: req.query.repo,
        number: req.query.number,
      });
      if (!pathResult.success) {
        return res.status(400).json({ error: 'Invalid path parameters', details: pathResult.error.flatten().fieldErrors });
      }

      const { owner, repo, number } = pathResult.data;
      const repoFull = `${owner}/${repo}`;
      const issueNumber = parseInt(number, 10);

      try {
        switch (req.method) {
          case 'GET': {
            // D-04: always 200 — return default empty state if no record exists, never 404
            const record = await getTriageRecord(repoFull, issueNumber);
            return res.json(
              record ?? {
                isTriaged: false,
                priority: null,
                githubCommentId: null,
                commentStatus: null,
              }
            );
          }

          case 'PUT': {
            // D-05: upsert semantics — create if absent, update if present
            const bodyResult = triagePutBodySchema.safeParse(req.body);
            if (!bodyResult.success) {
              return res.status(400).json({
                error: 'Invalid input',
                details: bodyResult.error.flatten().fieldErrors,
              });
            }
            const record = await upsertTriageRecord(repoFull, issueNumber, bodyResult.data);
            return res.json(record);
          }

          default:
            return res.status(405).json({ error: 'Method not allowed' });
        }
      } catch (error: any) {
        res.status(500).json({ error: 'Internal server error' });
      }
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → triage route | Untrusted: owner, repo, number in path params; isTriaged and priority in PUT body |
| Client → issues route | Untrusted: labels and assignee query params passed through to GitHub API |
| triage route → GitHub | No direct GitHub calls from triage route — trust boundary N/A for this route |
| issues route → GitHub API | labels/assignee values originate from client request, forwarded to GitHub via URLSearchParams |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Spoofing | GET/PUT `/api/triage/*` unauthenticated access | mitigate | `authenticateRequest()` called at top of handler per D-06; returns early if null, sending 401 automatically |
| T-03-02 | Tampering | `owner`/`repo` path params containing special chars | mitigate | `githubOwnerRepoSchema` validates both against `/^[a-zA-Z0-9_.-]+$/` regex; invalid values return 400 before any DB access |
| T-03-03 | Tampering | `number` path param containing non-integer input | mitigate | `triagePathSchema` extends with `z.string().regex(/^\d+$/)` before `parseInt`; invalid values return 400 |
| T-03-04 | Tampering | `priority` PUT body value injection | mitigate | `z.enum(['critical','high','medium','low']).nullable()` enforces enum constraint; anything else returns 400 |
| T-03-05 | Tampering | `labels` query param forwarded to GitHub | accept | Passed via URLSearchParams.set() which URL-encodes the value; worst case GitHub returns 422 which is handled by `!response.ok` branch |
| T-03-06 | Elevation of Privilege | Accessing other users' triage records | accept | Triage records are per (repo, issueNumber) not per user — intentional: all authenticated team members share triage state (TRIAGE-03 requirement) |
| T-03-07 | Denial of Service | `issue_triage` table unbounded growth | accept | Each (repo, issueNumber) pair is unique due to composite PK; one row per issue per repo — bounded by GitHub issue count |
</threat_model>
